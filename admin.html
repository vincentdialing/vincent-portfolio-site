Database Setup Helper

If you want to use your public Anon Key but are getting permissions errors, you must run these commands in your Supabase SQL Editor to enable write operations:

Enable Write Policies (SQL) 
-- 1. Add Custom Pill Badge support to Services
ALTER TABLE portfolio_services ADD COLUMN IF NOT EXISTS badge TEXT;

-- 2. Enable insert/update/delete for Anon Key on all tables
-- portfolio_services
CREATE POLICY "Allow anon insert" ON portfolio_services FOR INSERT WITH CHECK (true);
CREATE POLICY "Allow anon update" ON portfolio_services FOR UPDATE USING (true) WITH CHECK (true);
CREATE POLICY "Allow anon delete" ON portfolio_services FOR DELETE USING (true);

-- portfolio_projects
CREATE POLICY "Allow anon insert" ON portfolio_projects FOR INSERT WITH CHECK (true);
CREATE POLICY "Allow anon update" ON portfolio_projects FOR UPDATE USING (true) WITH CHECK (true);
CREATE POLICY "Allow anon delete" ON portfolio_projects FOR DELETE USING (true);

-- portfolio_project_images
CREATE POLICY "Allow anon insert" ON portfolio_project_images FOR INSERT WITH CHECK (true);
CREATE POLICY "Allow anon update" ON portfolio_project_images FOR UPDATE USING (true) WITH CHECK (true);
CREATE POLICY "Allow anon delete" ON portfolio_project_images FOR DELETE USING (true);

-- location_card
CREATE POLICY "Allow anon insert" ON location_card FOR INSERT WITH CHECK (true);
CREATE POLICY "Allow anon update" ON location_card FOR UPDATE USING (true) WITH CHECK (true);
CREATE POLICY "Allow anon delete" ON location_card FOR DELETE USING (true);

-- community_cards
CREATE POLICY "Allow anon insert" ON community_cards FOR INSERT WITH CHECK (true);
CREATE POLICY "Allow anon update" ON community_cards FOR UPDATE USING (true) WITH CHECK (true);
CREATE POLICY "Allow anon delete" ON community_cards FOR DELETE USING (true);

-- brands
CREATE POLICY "Allow anon insert" ON brands FOR INSERT WITH CHECK (true);
CREATE POLICY "Allow anon update" ON brands FOR UPDATE USING (true) WITH CHECK (true);
CREATE POLICY "Allow anon delete" ON brands FOR DELETE USING (true);

-- 3. Enable Storage uploads/updates/deletes for the 'portfolio' bucket (run to allow files upload via client fallback)
-- CREATE POLICY "Allow anon uploads to portfolio bucket" ON storage.objects FOR INSERT TO public WITH CHECK (bucket_id = 'portfolio');
-- CREATE POLICY "Allow anon updates to portfolio bucket" ON storage.objects FOR UPDATE TO public USING (bucket_id = 'portfolio') WITH CHECK (bucket_id = 'portfolio');
-- CREATE POLICY "Allow anon deletes from portfolio bucket" ON storage.objects FOR DELETE TO public USING (bucket_id = 'portfolio');
-- CREATE POLICY "Allow public select from portfolio bucket" ON storage.objects FOR SELECT TO public USING (bucket_id = 'portfolio');

-- 4. Alternatively, disable Row Level Security (RLS) entirely:
